<script>
'use strict';
const SK = 'inv_pro';
const EMOJI = {Pintura:'🖌️','Herramienta General':'🔨',Eléctrico:'⚡',Medición:'📋',Plomería:'🔧',Carpintería:'🪚',Seguridad:'🥾',Jardinería:'🌿',Limpieza:'🧹',_:'🔧'};
const de = d => EMOJI[d]||EMOJI._;
const esc = s => String(s||'').replace(/&/g,'&').replace(/</g,'<').replace(/>/g,'>').replace(/\"/g,'"');




let items = (()=>{ try{ const r=localStorage.getItem(SK); return r?JSON.parse(r):seed(); }catch{return seed();} })();
let filtDep = '';
let pendImg = null;
let editId = null;




function seed(){
  return [
    {id:'1',cod:'PNT-001',nom:'Brocha Pretul 1½\"',dep:'Pintura',qty:1,dsc:'Aplicación de pintura.',img:null},
    {id:'2',cod:'HTG-001',nom:'Mazo de Hule',dep:'Herramienta General',qty:1,dsc:'Para ensamble sin marcar superficies.',img:null}
  ];
}




function save(){ try{localStorage.setItem(SK,JSON.stringify(items));}catch(e){showToast('Error al guardar','err');} }




function renderList(){
  const bus = document.getElementById('buscar').value.toLowerCase().trim();
  const lista = document.getElementById('lista');
  lista.innerHTML = '';
  
  const filtrados = items.filter(i => {
    const matchBus = i.nom.toLowerCase().includes(bus) || i.cod.toLowerCase().includes(bus) || i.dsc.toLowerCase().includes(bus);
    const matchDep = !filtDep || i.dep === filtDep;
    return matchBus && matchDep;
  });




  if(filtrados.length === 0) {
    document.getElementById('empty').style.display = 'block';
  } else {
    document.getElementById('empty').style.display = 'none';
    filtrados.forEach(i => {
      const div = document.createElement('div');
      div.className = 'icard';
      div.innerHTML = `
        <div class="ithumb" onclick="viewImg('${i.id}')">
          ${i.img ? `<img src="${i.img}">` : de(i.dep)}
          ${i.img ? '<div class="tho">👁️ ver</div>' : ''}
        </div>
        <div class="iinfo" onclick="editItem('${i.id}')">
          <div class="icode">${esc(i.cod)}</div>
          <div class="iname">${esc(i.nom)}</div>
          <div class="idepto">${de(i.dep)} ${esc(i.dep)}</div>
        </div>
        <div class="iright">
          <div class="qctl">
            <button class="qbtn" onclick="event.stopPropagation(); modQty('${i.id}', -1)">-</button>
            <div class="qval">${i.qty}</div>
            <button class="qbtn" onclick="event.stopPropagation(); modQty('${i.id}', 1)">+</button>
          </div>
          <button class="bdel" onclick="event.stopPropagation(); delItem('${i.id}'">🗑️ Eliminar</button>
        </div>
      `;
      lista.appendChild(div);
    });
  }
  updateStats();
  renderChips();
  updateDatalist();
}




function updateStats(){
  const tot = items.length;
  const deps = [...new Set(items.map(i=>i.dep))].length;
  const uds = items.reduce((acc,i)=>acc+parseInt(i.qty||0), 0);
  document.getElementById('stot').textContent = tot;
  document.getElementById('sdep').textContent = deps;
  document.getElementById('sqty').textContent = uds;
  document.getElementById('cpill').textContent = `${tot} items`;
}




function renderChips(){
  const chips = document.getElementById('chips');
  const todosDeps = [...new Set(items.map(i=>i.dep))].filter(Boolean);
  chips.innerHTML = `<div class="chip ${!filtDep?'on':''} " onclick="setFilt('')">Todos</div>`;
  todosDeps.forEach(d => {
    chips.innerHTML += `<div class="chip ${filtDep===d?'on':''} " onclick="setFilt('${d}')">${de(d)} ${esc(d)}</div>`;
  });
}




function updateDatalist(){
  const dl = document.getElementById('dlist');
  const todosDeps = [...new Set(items.map(i=>i.dep))].filter(Boolean);
  dl.innerHTML = '';
  const predefinidos = ['Pintura', 'Herramienta General', 'Eléctrico', 'Medición', 'Plomería', 'Carpintería'];
  const combinados = [...new Set([...predefinidos, ...todosDeps])];
  combinados.forEach(d => {
    dl.innerHTML += `<option value="${esc(d)}">`;
  });
}




function setFilt(d){ filtDep = d; renderList(); }
function toggleFilter(){ document.getElementById('fpanel').classList.toggle('open'); document.getElementById('bfilt').classList.toggle('on'); }
function toggleClear(){ document.getElementById('bclr').style.display = document.getElementById('buscar').value ? 'block' : 'none'; }
function clearSearch(){ document.getElementById('buscar').value = ''; toggleClear(); renderList(); }




function openModal(){
  editId = null;
  document.getElementById('mtit').textContent = '➕ Nueva Herramienta';
  document.getElementById('fCod').value = '';
  document.getElementById('fQty').value = '1';
  document.getElementById('fNom').value = '';
  document.getElementById('fDep').value = '';
  document.getElementById('fDsc').value = '';
  removePhoto();
  document.getElementById('ovl').classList.add('open');
}




function closeModal(){ document.getElementById('ovl').classList.remove('open'); }
function handleOvl(e){ if(e.target.id === 'ovl') closeModal(); }




const handleImg = e => {
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onloadend = () => {
    pendImg = reader.result;
    document.getElementById('fpreviewImg').src = pendImg;
    document.getElementById('fpick').classList.add('hide');
    document.getElementById('fpreview').classList.add('show');
  };
  reader.readAsDataURL(file);
};
document.getElementById('_fc').addEventListener('change', handleImg);
document.getElementById('_fg').addEventListener('change', handleImg);




function removePhoto(){
  pendImg = null;
  document.getElementById('fpreviewImg').src = '';
  document
